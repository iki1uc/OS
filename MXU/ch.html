// ============================================================ // MXU_CH · Channel‑Header · Verbesserte Version // ============================================================ export const MXU_CH = { // Dynamische Eigenschaften id: "MXU-CH", version: "2.0.1", mode: "CHANNEL", axis: "B°", state: "INIT", timestamp: null, log: [], // ------------------------------------------------------------ // INIT · Startet das Modul // ------------------------------------------------------------ init(config = {}) { // Dynamische Achse / Modus / Version this.axis = config.axis || this.axis; this.mode = config.mode || this.mode; this.version = config.version || this.version; // Zeitstempel this.timestamp = Date.now(); this.state = "OK"; // Log-Eintrag this._addLog("INIT", `MXU_CH gestartet (Axis=${this.axis}, Mode=${this.mode})`); return { id: this.id, mode: this.mode, state: this.state, axis: this.axis, version: this.version, timestamp: this.timestamp }; }, // ------------------------------------------------------------ // LOGGING · interne Log-Funktion // ------------------------------------------------------------ _addLog(type, msg) { this.log.push({ type, msg, time: new Date().toISOString() }); }, // ------------------------------------------------------------ // STATUS · gibt den aktuellen Zustand zurück // ------------------------------------------------------------ status() { return { id: this.id, state: this.state, axis: this.axis, version: this.version, logCount: this.log.length, lastEvent: this.log[this.log.length - 1] || null }; }, // ------------------------------------------------------------ // ERROR · Fehlerbehandlung // ------------------------------------------------------------ error(msg) { this.state = "ERROR"; this._addLog("ERROR", msg); return { state: "ERROR", msg }; }, // ------------------------------------------------------------ // SYNC · wird von SLI.attach() überschrieben // ------------------------------------------------------------ sync() { this._addLog("SYNC", "SLI Sync ausgeführt"); return { sync: true, time: Date.now() }; } };
